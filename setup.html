みんなのウォール セットアップ

この作業は1回だけ。共有ウォール（みんなの投稿が見える機能）を有効にします。
所要 約1分。個人ノートはこの作業なしでも使えます。

1Supabase を開く

  1. この SQL エディタを開く →（Re:alizeと同じプロジェクト）
  2. ログインを求められたら、いつものアカウントで入る

2下の呪文をコピーして貼り付け → RUN

「Copy」を押してSQLエディタに貼り、右下の RUN（またはCmd+Enter）を押すだけ。

-- Merone 北海道 STUDY TRIP : 共有ウォール
create table if not exists hokkaido_notes (
  id uuid primary key default gen_random_uuid(),
  device_id text not null,
  display_name text not null,
  mission_id text not null,
  choice text,
  feeling text,
  body text,
  created_at timestamptz default now()
);
create table if not exists hokkaido_reactions (
  id uuid primary key default gen_random_uuid(),
  note_id uuid not null references hokkaido_notes(id) on delete cascade,
  device_id text not null,
  emoji text not null,
  created_at timestamptz default now(),
  unique (note_id, device_id, emoji)
);
alter table hokkaido_notes enable row level security;
alter table hokkaido_reactions enable row level security;
create policy "n_sel" on hokkaido_notes for select to anon using (true);
create policy "n_ins" on hokkaido_notes for insert to anon with check (true);
create policy "n_del" on hokkaido_notes for delete to anon using (true);
create policy "r_sel" on hokkaido_reactions for select to anon using (true);
create policy "r_ins" on hokkaido_reactions for insert to anon with check (true);
create policy "r_del" on hokkaido_reactions for delete to anon using (true);
